name: Build D6300 firmware

on:
  workflow_dispatch:
  push:
    branches: [ main ]

jobs:
  build:
    runs-on: ubuntu-22.04
    timeout-minutes: 350
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Free disk space
        run: |
          sudo rm -rf /usr/share/dotnet /usr/local/lib/android /opt/ghc || true
          df -h

      - name: Install build dependencies
        run: |
          sudo apt-get update
          sudo apt-get install -y build-essential clang flex bison g++ gawk \
            gcc-multilib g++-multilib gettext git libncurses5-dev libssl-dev \
            python3 python3-setuptools rsync swig unzip zlib1g-dev file wget \
            libelf-dev zstd python3-distutils libncurses-dev

      - name: Clone OpenWrt
        run: |
          git clone --depth 1 https://git.openwrt.org/openwrt/openwrt.git openwrt-src
          cd openwrt-src
          git log --oneline -1 | tee ../openwrt-commit.txt
          git config user.email "ci@d6300.local"
          git config user.name "D6300 CI"

      - name: Apply D6300 patches
        run: |
          cd openwrt-src
          for p in ../openwrt/patches/0001-*.patch \
                   ../openwrt/patches/0002-*.patch \
                   ../openwrt/patches/0003-*.patch; do
            echo "== applying $p =="
            git am "$p"
          done
          git log --oneline -4

      - name: Configure target
        run: |
          cd openwrt-src
          cat >> .config <<'CONFIGEOF'
          CONFIG_TARGET_bmips=y
          CONFIG_TARGET_bmips_bcm6362=y
          CONFIG_TARGET_bmips_bcm6362_DEVICE_netgear_d6300=y
          CONFIGEOF
          make defconfig
          grep -q "CONFIG_TARGET_bmips_bcm6362_DEVICE_netgear_d6300=y" .config \
            && echo "TARGET OK: netgear_d6300"

      - name: Inject cferam into bcm63xx-cfe package
        run: |
          cd openwrt-src
          make package/kernel/bcm63xx-cfe/prepare V=s
          SRC=$(find build_dir -maxdepth 4 -type d -name "bcm63xx-cfe-*" | head -n1)
          echo "package source: $SRC"
          mkdir -p "$SRC/netgear,d6300"
          cp ../openwrt/bcm63xx-cfe/netgear,d6300/cferam.000 "$SRC/netgear,d6300/"
          ls -l "$SRC/netgear,d6300/"

      - name: Download sources
        run: |
          cd openwrt-src
          make download -j8 V=s || make download -j1 V=s

      - name: Build firmware
        run: |
          cd openwrt-src
          df -h
          make -j$(nproc) V=s 2>&1 | tail -n 300

      - name: Collect images
        if: always()
        run: |
          mkdir -p out
          cp -v openwrt-src/bin/targets/bmips/bcm6362/* out/ 2>/dev/null || true
          cp openwrt-commit.txt out/ 2>/dev/null || true
          ls -la out

      - name: Upload firmware artifacts
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: d6300-openwrt-images
          path: out/
          if-no-files-found: warn
          retention-days: 30
